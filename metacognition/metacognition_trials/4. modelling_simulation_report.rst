    R0 = 40

    predation_rate = 0.02
    predator_growth_rate = 0.01
    predator_death_rate = 0.1
    dt = 1
    n_steps = 100
    
    # Initialize populations
    R = [R0]

    # Euler method simulation
    for t in range(n_steps):

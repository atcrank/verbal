========================================================================
Trial 7: Iterative Planning (Task Decomposer)
========================================================================

This test verifies the agent's ability to iteratively plan and execute a dynamic JSON queue of tasks, 
specifically verifying that it processes each task sequentially without trailing off.

    >>> import contextlib, io, sys
    >>> import uuid
    >>> from llm_api.models import User
    >>> from metacognition.models import CognitiveBlueprint
    >>> from metacognition.tasks import run_blueprint
    >>>
    >>> user, _ = User.objects.get_or_create(username="test_user", defaults={"is_staff": True, "is_superuser": True})
    >>> if not (user.is_staff and user.is_superuser):
    ...     user.is_staff = True
    ...     user.is_superuser = True
    ...     user.save(update_fields=["is_staff", "is_superuser"])
    >>> from metacognition.seed import seed_task_decomposer
    >>> from metacognition.models import ReasoningStep, ResponseSchema, ToolDefinition
    >>> seed_task_decomposer(CognitiveBlueprint, ReasoningStep, ResponseSchema, ToolDefinition)
    >>> bp = CognitiveBlueprint.objects.filter(name__startswith="Task Decomposer").first()
    >>> cid = str(uuid.uuid4())
    >>> from llm_api.models import Conversation
    >>> _ = Conversation.objects.create(id=cid, user=user)
    
    >>> prompt = (
    ...     "We need to analyze our Django project. Task 1: List all models in `metacognition/models.py`. "
    ...     "Task 2: List all functions in `metacognition/actions.py`. "
    ...     "Task 3: Provide a brief summary of how they relate."
    ... )
    
    >>> with contextlib.redirect_stdout(io.StringIO()) as captured_out:
    ...     result = run_blueprint(bp.id, prompt, conversation_id=cid, user_id=user.id)
    
    >>> conv = Conversation.objects.get(id=cid)
    >>> state_tree = conv.state_tree or {}
    >>> if result:
    ...     result["stdout"] = captured_out.getvalue()
    ...     tasks_map = state_tree.get("tasks", {}) if isinstance(state_tree, dict) else {}
    ...     bad_completed = [tid for tid, info in tasks_map.items() if isinstance(info, dict) and info.get("status") == "COMPLETED" and ("not found" in str(info.get("resolution", "")).lower() or "error" in str(info.get("resolution", "")).lower())]
    ...     final_str = str(result.get("final_response", ""))
    ...     expected_outcome = (
    ...         "Decompose into >= 3 tasks (models in metacognition/models.py, functions in metacognition/actions.py, "
    ...         "and relationship summary), execute sub-blueprints without falsely marking errored or skipped tasks COMPLETED, "
    ...         "and compile a factual project overview."
    ...     )
    ...     completed_cnt = sum(1 for info in tasks_map.values() if isinstance(info, dict) and info.get("status") == "COMPLETED")
    ...     checks = [
    ...         {"label": "decomposed into >= 3 tasks", "passed": len(tasks_map) >= 3, "detail": f"tasks={len(tasks_map)}"},
    ...         {"label": "no failed/errored task falsely marked COMPLETED", "passed": len(bad_completed) == 0, "detail": f"bad_completed={bad_completed}"},
    ...         {"label": "all 3 tasks actually executed & COMPLETED without asking user for code", "passed": completed_cnt >= 3 and "please share" not in final_str.lower() and "provide the contents" not in final_str.lower() and "not available" not in final_str.lower(), "detail": f"completed={completed_cnt}/{len(tasks_map)}"},
    ...     ]
    ...     report_doctest_run("7. iterative_planning", [prompt], result, expected_outcome=expected_outcome, checks=checks)

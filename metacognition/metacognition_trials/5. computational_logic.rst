========================================
Trial 5: Computational Logic (PyAgrum)
========================================

This test verifies the agent's ability to construct and solve a strategic influence diagram 
using the `pyAgrum` library inside the sandbox environment.

    >>> import contextlib, io, sys
    >>> import uuid
    >>> from llm_api.models import User
    >>> from metacognition.models import CognitiveBlueprint
    >>> from metacognition.tasks import run_blueprint
    >>>
    >>> user, _ = User.objects.get_or_create(username="test_user", defaults={"is_staff": True, "is_superuser": True})
    >>> if not (user.is_staff and user.is_superuser):
    ...     user.is_staff = True
    ...     user.is_superuser = True
    ...     user.save(update_fields=["is_staff", "is_superuser"])
    >>> bp = CognitiveBlueprint.objects.filter(name__startswith="Escalation of Effort").first()
    >>> cid = str(uuid.uuid4())
    >>> from llm_api.models import Conversation
    >>> _ = Conversation.objects.create(id=cid, user=user)
    
    >>> prompt = (
    ...     "A company is deciding whether to launch a new product. "
    ...     "The market demand can be 'High' or 'Low' with 50% probability each. "
    ...     "The decision is to 'Launch' or 'NotLaunch'. "
    ...     "If they launch and demand is High, the profit is $100. "
    ...     "If they launch and demand is Low, the profit is $-50. "
    ...     "If they don't launch, the profit is $0 regardless. "
    ...     "Please build an Influence Diagram to find the optimal decision policy and expected utility."
    ... )
    
    >>> with contextlib.redirect_stdout(io.StringIO()) as captured_out:
    ...     result = run_blueprint(bp.id, prompt, conversation_id=cid, user_id=user.id)
    
    >>> if result:
    ...     result["stdout"] = captured_out.getvalue()
    ...     expected_outcome = (
    ...         "Solve the two-action decision problem in Python (no SyntaxError): Launch yields "
    ...         "EU = 0.5*(100) + 0.5*(-50) = $25 vs NotLaunch = $0. Optimal policy: Launch; Expected Utility: $25."
    ...     )
    ...     final_str = str(result.get("final_response", ""))
    ...     monologue = result.get("internal_monologue", [])
    ...     ran_sandbox = any("Sandbox Execution Succeeded" in str(m.get("tool_result", "")) or "Sandbox Execution Succeeded" in str(m.get("output", "")) for m in monologue)
    ...     checks = [
    ...         {"label": "sandbox script executed without syntax error", "passed": ran_sandbox, "detail": "RC=0"},
    ...         {"label": "optimal policy 'Launch' stated", "passed": "launch" in final_str.lower(), "detail": ""},
    ...         {"label": "expected utility $25 computed", "passed": "25" in final_str, "detail": ""},
    ...     ]
    ...     report_doctest_run("5. computational_logic", [prompt], result, expected_outcome=expected_outcome, checks=checks)
    
    >>> assert "error" not in result, f"Execution failed: {result.get('error')}"
    >>> assert result.get("final_response") != "No output.", "No response generated."
    >>> assert len(result.get("internal_monologue", [])) > 0, "Execution produced empty monologue."

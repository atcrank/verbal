Scenario 1.3: Monte Carlo Simulation

This advanced scenario tests mathematical modeling, the use of external libraries (numpy), and multi-step reasoning. It requires the AI to first draft a strategy before writing code.

Execution

We target the canonical ResearchEvaluation pipeline (Research, StrategicPlan, Sandboxed Execution, and Evaluator Critique).

>>> from metacognition.tasks import run_blueprint
>>> from metacognition.models import CognitiveBlueprint
>>> from django.contrib.auth.models import User
>>> user, _ = User.objects.get_or_create(username="test_user", defaults={"is_staff": True, "is_superuser": True})
>>> if not (user.is_staff and user.is_superuser):
...     user.is_staff = True
...     user.is_superuser = True
...     user.save(update_fields=["is_staff", "is_superuser"])
>>> bp = CognitiveBlueprint.objects.filter(name__startswith="ResearchEvaluation").first()

>>> prompt = (
...     "If we toss a fair coin to decide whether (HEADS) to roll a D6 (six-sided dice) "
...     "or (TAILS) a D20 (20-sided dice), I want to use a simple monte carlo model to compute "
...     "the expected average value and the standard deviation for a reasonably informative number of trials. "
...     "Write the simulation script, run it, and tell me the mean and standard deviation."
... )

>>> import contextlib, io, re
>>> with contextlib.redirect_stdout(io.StringIO()) as captured_out:
...     result = run_blueprint(bp.id, prompt, user_id=user.id)
>>> result["stdout"] = captured_out.getvalue()

>>> expected_outcome = (
...     "Run a Monte Carlo Python simulation in the sandbox (e.g. >= 10,000 trials) choosing D6 or D20 with "
...     "50% probability each. Analytical expected value is 7.0 and population standard deviation is ~5.51."
... )
>>> final_str = result.get("final_response", "")
>>> monologue = result.get("internal_monologue", [])
>>> ran_sandbox = any("Sandbox Execution Succeeded" in str(m.get("tool_result", "")) or "Sandbox Execution Succeeded" in str(m.get("output", "")) for m in monologue)
>>> nums = [float(x) for x in re.findall(r"\b\d+(?:\.\d+)?\b", final_str)]
>>> has_mean = any(6.5 <= n <= 7.5 for n in nums)
>>> has_std = any(4.8 <= n <= 6.2 for n in nums)
>>> checks = [
...     {"label": "sandbox simulation executed (RC=0)", "passed": ran_sandbox, "detail": "python_sandbox output recorded"},
...     {"label": "mean near analytical 7.0", "passed": has_mean, "detail": f"numbers found: {nums[:10]}"},
...     {"label": "std dev near analytical 5.51", "passed": has_std, "detail": f"numbers found: {nums[:10]}"},
... ]
>>> report_doctest_run("2. design_compute", prompt, result, expected_outcome=expected_outcome, checks=checks)

>>> assert "error" not in result, f"Execution failed: {result.get('error')}"
>>> assert len(monologue) > 0 and bool(final_str), "Blueprint produced empty output"
>>> print(f"- complete attempt: {1 if final_str else 0}/1")
- complete attempt: 1/1

import random

# Configuration
NUM_TRIALS = 100000  # A large number for better accuracy

rolls = []

# Simulation loop
for _ in range(NUM_TRIALS):
    # Flip a coin to decide the die type (0 for D6, 1 for D20)
    die_type = random.randint(0, 1)

    if die_type == 0:  # Heads: Roll a D6
        roll_value = random.randint(1, 6)
    else:  # Tails: Roll a D20
        roll_value = random.randint(1, 20)

    rolls.append(roll_value)

# Calculation
if rolls:
    # Calculate Mean
    mean = sum(rolls) / len(rolls)

    # Calculate Standard Deviation
    # Variance = sum((x - mean)^2 for x in rolls) / N
    variance = sum((x - mean) ** 2 for x in rolls) / len(rolls)
    std_dev = variance ** 0.5

    print(f"Number of trials: {len(rolls)}")
    print(f"Estimated Mean: {mean:.4f}")
    print(f"Estimated Standard Deviation: {std_dev:.4f}")
else:
    print("No rolls were recorded.")

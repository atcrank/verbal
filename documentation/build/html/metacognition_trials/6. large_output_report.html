import pandas as pd
import numpy as np
import random
import csv

# --- Script 1: Generate the dataset and save it ---

NUM_EMPLOYEES = 1000
DEPARTMENTS = ['Engineering', 'Marketing', 'Sales', 'HR', 'Finance']

# Generate data
data = {
    'Name': [f'Employee_{i+1}' for i in range(NUM_EMPLOYEES)],
    'Department': np.random.choice(DEPARTMENTS, NUM_EMPLOYEES),
    'Salary': np.random.randint(40000, 120000, NUM_EMPLOYEES),
    'Performance Score': np.random.randint(1, 6, NUM_EMPLOYEES)
}

df = pd.DataFrame(data)

# Save to CSV
csv_filename = 'employees.csv'
df.to_csv(csv_filename, index=False)
print(f"Successfully generated and saved synthetic dataset to {csv_filename}")
print("\n--- Starting Script 2: Statistical Analysis ---\n")

# --- Script 2: Read the CSV and calculate statistics ---

# Read the CSV
try:
    df_analysis = pd.read_csv(csv_filename)
    print("Successfully loaded the dataset for analysis.")

    # Calculate mean salary per department
    salary_summary = df_analysis.groupby('Department')['Salary'].mean().sort_values(ascending=False)

    print("\n--- Statistical Summary: Mean Salary per Department ---")
    print(salary_summary.to_string())

except FileNotFoundError:
    print(f"Error: The file {csv_filename} was not found.")
except Exception as e:
    print(f"An error occurred during analysis: {e}")

    for line in content.splitlines():
        if line.strip().startswith("class ") and "models." in line:
            # A simple heuristic to capture class definitions that look like models
            models.append(line.strip())

    if models:
        print("--- Models in metacognition/models.py --- ")
        for model in models:
            print(model)
    else:
        print("Could not find models using simple string matching.")
